    function renderPalette() {
      const headings = ["Family"].concat(colourShades);
      const gridItems = headings.map(function(heading) {
        const element = document.createElement("span");
        element.className = "palette-column-heading";
        element.textContent = heading;
        if (heading === "Family") {
          element.classList.add("palette-row-heading");
        }
        return element;
      });

      colourFamilies.forEach(function(family) {
        const familyLabel = document.createElement("span");
        familyLabel.className = "palette-row-heading";
        familyLabel.textContent =
          family.charAt(0).toUpperCase() + family.slice(1);
        gridItems.push(familyLabel);

        colourShades.forEach(function(shade) {
          const colourName = family + "-" + shade;
          const swatch = document.createElement("button");
          swatch.className = "palette-swatch";
          swatch.type = "button";
          swatch.dataset.colour = colourName;
          swatch.setAttribute("aria-label", colourName);
          swatch.setAttribute("aria-pressed", "false");
          swatch.title = colourName;
          gridItems.push(swatch);
        });
      });

      paletteGrid.replaceChildren.apply(paletteGrid, gridItems);

      const rules = paletteColours.map(function(colourName) {
        return '.palette-swatch[data-colour="' + colourName + '"] {' +
          " background-color: var(--color-" + colourName + "); }";
      });
      swatchStyles.textContent = rules.join("\n");
    }

    function readSettings() {
      settings.dark = darkCheckbox.checked;
      settings.gradient = gradientCheckbox.checked;
      settings.centre = centreCheckbox.checked;
      settings.minimumHeight = heightCheckbox.checked;
      settings.showSubheading = subheadingCheckbox.checked;
      settings.h1Serif = h1SerifCheckbox.checked;
      settings.h2Serif = h2SerifCheckbox.checked;
      settings.h1LetterSpacing =
        Number(h1LetterSpacingSlider.value);
      settings.h2LetterSpacing =
        Number(h2LetterSpacingSlider.value);
      settings.gradientDirection = gradientDirection.value;
    }

    function getHeroClassNames() {
      const classNames = ["hero", settings.dark ? "hero-dark" : "hero-light"];

      if (settings.centre) {
        classNames.push("hero-centre");
      }
      if (settings.minimumHeight) {
        classNames.push("hero-min-height");
      }
      if (settings.h1Serif) {
        classNames.push("hero-h1-serif");
      }
      if (settings.h2Serif) {
        classNames.push("hero-h2-serif");
      }

      return classNames.join(" ");
    }

    function formatLetterSpacing(value) {
      return value.toFixed(3) + "em";
    }

    function updateLetterSpacingStyles() {
      const h1Spacing =
        formatLetterSpacing(settings.h1LetterSpacing);
      const h2Spacing =
        formatLetterSpacing(settings.h2LetterSpacing);

      h1SpacingValue.textContent = h1Spacing;
      h2SpacingValue.textContent = h2Spacing;
      spacingStyles.textContent = `
    .hero h1 {
      letter-spacing: ${h1Spacing};
    }

      letter-spacing: ${h2Spacing};
    }
`;
    }

      const backgroundRule = settings.gradient

        var(--color-${settings.gradientEndColour})
      );`

      colourStyles.textContent = `
    .hero {
      background-color: var(--color-${settings.backgroundColour});
      ${backgroundRule}
    }

    .hero h1 {
      color: var(--color-${settings.h1Colour});
    }

    .hero h2 {
      color: var(--color-${settings.h2Colour});
    }
`;
    }

    function getColourSetting(target) {
      const colourSettings = {
        background: "backgroundColour",
        gradientStart: "gradientStartColour",
        gradientEnd: "gradientEndColour",
        h1: "h1Colour",
        h2: "h2Colour"

    function getColourTargetName(target) {
      const targetNames = {
        background: "Background",
        gradientStart: "Start colour",
        gradientEnd: "End colour",

      return targetNames[target];

    function updatePaletteTargets() {
      backgroundTarget.hidden = settings.gradient;
      gradientStartTarget.hidden = !settings.gradient;
      gradientEndTarget.hidden = !settings.gradient;
      gradientControls.hidden = !settings.gradient;

        targetButtons[settings.activeColourTarget].hidden;
      const activeTargetDisabled =
        targetButtons[settings.activeColourTarget].disabled;

      if (activeTargetHidden || activeTargetDisabled) {
        settings.activeColourTarget =
          settings.gradient ? "gradientStart" : "background";
      }
    }

    function updatePaletteSelection() {
      const selectedColour =
        settings[getColourSetting(settings.activeColourTarget)];
      const targetName =
        getColourTargetName(settings.activeColourTarget);

      paletteSelection.textContent =
        targetName + " colour: " + selectedColour;

      Object.keys(targetButtons).forEach(function(target) {
        targetButtons[target].setAttribute(
          "aria-pressed",
          String(settings.activeColourTarget === target)
        );
      });

      paletteGrid.querySelectorAll(".palette-swatch").forEach(
        function(swatch) {
          swatch.setAttribute(
            "aria-pressed",

        }

        .forEach(function(swatch) {

    function resetPresetColours() {
      settings.dark = darkCheckbox.checked;

      if (!paletteColours.includes(colourName)) {

    function handlePlainTextPaste(event) {
      event.preventDefault();
      const pastedText = event.clipboardData
        ? event.clipboardData.getData("text/plain")
        : "";
      insertPlainTextAtCursor(pastedText);
    }

    function insertPlainTextAtCursor(text) {
      const selection = window.getSelection();
      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const range = selection.getRangeAt(0);
      const activeElement = document.activeElement;
      if (
        !activeElement ||
        !activeElement.contains(range.commonAncestorContainer)
      ) {
        return;
      }

      range.deleteContents();
      const textNode = document.createTextNode(text);
      range.insertNode(textNode);
      range.setStartAfter(textNode);
      range.collapse(true);
      selection.removeAllRanges();
      selection.addRange(range);
      activeElement.dispatchEvent(new Event("input", { bubbles: true }));
    }

    function preventNewLines(event) {
      if (event.key === "Enter") {
        event.preventDefault();
      }
    }

    function normaliseEditableHeading(heading) {
      if (!heading.querySelector("*")) {
        return;
      }

      const plainText = heading.textContent;
      const selection = window.getSelection();
      let cursorOffset = plainText.length;

      if (
        selection &&
        selection.rangeCount > 0 &&
        heading.contains(selection.anchorNode)
      ) {
        const beforeCursor = selection.getRangeAt(0).cloneRange();
        beforeCursor.selectNodeContents(heading);
        beforeCursor.setEnd(
          selection.anchorNode,
          selection.anchorOffset
        );
        cursorOffset = beforeCursor.toString().length;
      }

      heading.textContent = plainText;
      const textNode = heading.firstChild;
      if (selection && textNode) {
        const range = document.createRange();
        range.setStart(
          textNode,
          Math.min(cursorOffset, textNode.length)
        );
        range.collapse(true);
        selection.removeAllRanges();
        selection.addRange(range);
      }
    }

    function escapeHtml(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    function generateHtml() {
      const heroClasses = getHeroClassNames();
      const titleText = escapeHtml(heroTitle.textContent);
      const subtitleText = escapeHtml(heroSubheading.textContent);
      const subheadingMarkup = settings.showSubheading
        ? "\n        <h2>" + subtitleText + "</h2>"
        : "";
      const tailwindScript =
        '<script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>';

      return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Hero Section</title>
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

  ${tailwindScript}

  <style type="text/tailwindcss">${heroStyles.textContent}</style>

  <style>${spacingStyles.textContent}</style>

  <style>${colourStyles.textContent}</style>
</head>
<body>
  <header class="${heroClasses}">
    <div class="hero-content">
      <h1>${titleText}</h1>${subheadingMarkup}
    </div>
  </header>
</body>
</html>`;
    }

    function updateManualCopy() {
      if (!manualCopyPanel.hidden) {
        manualCopyTextarea.value = generateHtml();
      }
    }

    async function copyHtml() {
      readSettings();
      updateLetterSpacingStyles();
      updateColourStyles();

      const generatedHtml = generateHtml();
      copyStatus.textContent = "";

      try {
        if (!navigator.clipboard || !navigator.clipboard.writeText) {
          throw new Error("Clipboard API unavailable");
        }
        await navigator.clipboard.writeText(generatedHtml);
        copyStatus.textContent =
          "HTML copied—paste it into VS Code.";
        manualCopyPanel.hidden = true;
      } catch (error) {
        manualCopyPanel.hidden = false;
        manualCopyTextarea.value = generatedHtml;
        copyStatus.textContent =
          "Automatic copying is unavailable. " +
          "Select and copy the HTML below.";
        manualCopyTextarea.focus();
        manualCopyTextarea.select();
      }
    }
