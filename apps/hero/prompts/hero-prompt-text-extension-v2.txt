  <script>
    "use strict";

    // 1. Find the controls and editable headings.
    const darkCheckbox =
      document.getElementById("dark-checkbox");

    const centreCheckbox =
      document.getElementById("centre-checkbox");

    const heightCheckbox =
      document.getElementById("height-checkbox");

    const subheadingCheckbox =
      document.getElementById("subheading-checkbox");

    const heroPreview =
      document.getElementById("hero-preview");

    const heroTitle =
      document.getElementById("hero-title");

    const heroSubheading =
      document.getElementById("hero-subheading");

    const copyButton =
      document.getElementById("copy-button");

    const copyStatus =
      document.getElementById("copy-status");

    const manualCopyPanel =
      document.getElementById("manual-copy-panel");

    const manualCopyTextarea =
      document.getElementById("manual-copy-textarea");

    const heroStyles =
      document.getElementById("hero-styles");

    // 2. Store the four current checkbox settings.
    const settings = {
      dark: false,
      centre: false,
      minimumHeight: true,
      showSubheading: true
    };

    // Read the actual checked property of each checkbox.
    function readSettings() {
      settings.dark = darkCheckbox.checked;
      settings.centre = centreCheckbox.checked;
      settings.minimumHeight = heightCheckbox.checked;
      settings.showSubheading = subheadingCheckbox.checked;
    }

    // Build the same hero class list for preview and export.
    function getHeroClassNames() {
      const classNames = ["hero"];

      classNames.push(
        settings.dark ? "hero-dark" : "hero-light"
      );

      if (settings.centre) {
        classNames.push("hero-centre");
      }

      if (settings.minimumHeight) {
        classNames.push("hero-min-height");
      }

      return classNames.join(" ");
    }

    // 3. Apply settings without replacing editable elements.
    function updateHero() {
      readSettings();

      heroPreview.className = getHeroClassNames();

      // Hidden elements take no space or keyboard focus.
      heroSubheading.hidden = !settings.showSubheading;

      // Keep manual fallback HTML current, if displayed.
      updateManualCopy();
    }

    // Insert pasted content as plain text only.
    function handlePlainTextPaste(event) {
      event.preventDefault();

      const pastedText = event.clipboardData
        ? event.clipboardData.getData("text/plain")
        : "";

      insertPlainTextAtCursor(pastedText);
    }

    // Insert text at the current selection, without HTML.
    function insertPlainTextAtCursor(text) {
      const selection = window.getSelection();

      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const range = selection.getRangeAt(0);
      const activeElement = document.activeElement;

      if (
        !activeElement ||
        !activeElement.contains(range.commonAncestorContainer)
      ) {
        return;
      }

      range.deleteContents();

      const textNode = document.createTextNode(text);

      range.insertNode(textNode);

      // Put the cursor after the inserted text.
      range.setStartAfter(textNode);
      range.collapse(true);

      selection.removeAllRanges();
      selection.addRange(range);

      // Notify the editor that its text has changed.
      activeElement.dispatchEvent(
        new Event("input", { bubbles: true })
      );
    }

    // Keep the headings single-line in editing behaviour.
    // Long text still wraps naturally within the hero.
    function preventNewLines(event) {
      if (event.key === "Enter") {
        event.preventDefault();
      }
    }

    // Remove rich formatting introduced by other input
    // methods, while keeping the same heading element.
    function normaliseEditableHeading(heading) {
      if (heading.querySelector("*")) {
        const plainText = heading.textContent;
        const selection = window.getSelection();

        // Save the cursor's character position.
        let cursorOffset = plainText.length;

        if (
          selection &&
          selection.rangeCount > 0 &&
          heading.contains(selection.anchorNode)
        ) {
          const beforeCursor = selection
            .getRangeAt(0)
            .cloneRange();

          beforeCursor.selectNodeContents(heading);
          beforeCursor.setEnd(
            selection.anchorNode,
            selection.anchorOffset
          );

          cursorOffset = beforeCursor.toString().length;
        }

        heading.textContent = plainText;

        // Restore the cursor position.
        const textNode = heading.firstChild;

        if (selection && textNode) {
          const range = document.createRange();

          range.setStart(
            textNode,
            Math.min(cursorOffset, textNode.length)
          );

          range.collapse(true);
          selection.removeAllRanges();
          selection.addRange(range);
        }
      }
    }

    // 4. Escape heading text for safe HTML output.
    function escapeHtml(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    // 5. Create a complete standalone HTML document.
    function generateHtml() {
      const heroClasses = getHeroClassNames();

      const titleText = escapeHtml(heroTitle.textContent);
      const subtitleText = escapeHtml(
        heroSubheading.textContent
      );

      // The subheading is omitted when switched off.
      const subheadingMarkup = settings.showSubheading
        ? `\n        <h2>${subtitleText}</h2>`
        : "";

      // Reuse the exact shared hero CSS from the head.
      const exportedCss = heroStyles.textContent;

      // Escaping the closing script tag is important
      // inside the generator's JavaScript.
      const tailwindScript =
        '<script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>';

      return `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Hero Section</title>
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">

  ${tailwindScript}

  <style type="text/tailwindcss">${exportedCss}</style>
</head>
<body>
  <header class="${heroClasses}">
    <div class="hero-content">
      <h1>${titleText}</h1>${subheadingMarkup}
    </div>
  </header>
</body>
</html>`;
    }

    // Update the manual fallback after edits or changes.
    function updateManualCopy() {
      if (!manualCopyPanel.hidden) {
        manualCopyTextarea.value = generateHtml();
      }
    }

    // 6. Copy the generated document to the clipboard.
    async function copyHtml() {
      // Always read the latest settings and heading text.
      readSettings();

      const generatedHtml = generateHtml();

      copyStatus.textContent = "";

      try {
        if (
          !navigator.clipboard ||
          !navigator.clipboard.writeText
        ) {
          throw new Error("Clipboard API unavailable");
        }

        await navigator.clipboard.writeText(generatedHtml);

        // Only report success after copying succeeds.
        copyStatus.textContent =
          "HTML copied—paste it into VS Code.";

        manualCopyPanel.hidden = true;
      } catch (error) {
        // Graceful fallback for blocked clipboard access.
        manualCopyPanel.hidden = false;
        manualCopyTextarea.value = generatedHtml;

        copyStatus.textContent =
          "Automatic copying is unavailable. " +
          "Select and copy the HTML below.";

        manualCopyTextarea.focus();
        manualCopyTextarea.select();
      }
    }

    // 7. Connect the four controls to the preview.
    darkCheckbox.addEventListener("change", updateHero);
    centreCheckbox.addEventListener("change", updateHero);
    heightCheckbox.addEventListener("change", updateHero);
    subheadingCheckbox.addEventListener(
      "change",
      updateHero
    );

    // Connect text editing to the plain-text handlers.
    [heroTitle, heroSubheading].forEach(function(heading) {
      heading.addEventListener(
        "paste",
        handlePlainTextPaste
      );

      heading.addEventListener(
        "keydown",
        preventNewLines
      );

      heading.addEventListener("beforeinput", function(event) {
        if (
          event.inputType === "insertParagraph" ||
          event.inputType === "insertLineBreak"
        ) {
          event.preventDefault();
        }
      });

      heading.addEventListener("input", function() {
        normaliseEditableHeading(heading);
        updateManualCopy();
      });
    });

    copyButton.addEventListener("click", copyHtml);

    // 8. Initialise the preview using checkbox defaults.
    updateHero();
  </script>

</body>
</html>
